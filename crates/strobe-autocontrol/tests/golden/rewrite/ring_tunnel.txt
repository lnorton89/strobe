==== passes/st_clock.glsl
// ---- Strobe clock pass (generated) ----
// One float texel: x = iTime, y = last TIME, z = real dt, w = start frame + 1.
float strobe_rate() {
    return st_speed * mix(1.0, 0.3 + 2.0 * syn_BassLevel, st_audio_time);
}
vec4 renderMain() {
    vec4 prev = texelFetch(st_clock, ivec2(0, 0), 0);
    bool fresh = prev.w == 0.0 || TIME < prev.y;
    float dt = fresh ? 0.0 : TIME - prev.y;
    float clock = fresh ? TIME : prev.x + dt * strobe_rate();
    float start = fresh ? FRAMECOUNT + 1.0 : prev.w;
    return vec4(clock, TIME, dt, start);
}

==== main.glsl
// ---- Strobe import of Shadertoy Tst101 "Ring tunnel" by strobe-tests (generated) ----
// Pass: Image. Shim above the first marker; Shadertoy code below, unedited.
// Tab line 1 is file line 20 for Image.
vec2 strobe_fragCoord = vec2(0.0);
vec4 strobe_clock = texelFetch(st_clock, ivec2(0, 0), 0);
vec3 strobe_iResolution = vec3(RENDERSIZE, 1.0);
float strobe_iTime = strobe_clock.x;
void strobe_begin() {
    strobe_fragCoord = (gl_FragCoord.xy - 0.5 * RENDERSIZE) / st_zoom + 0.5 * RENDERSIZE;
}
vec4 strobe_end(vec4 c) {
    c.rgb *= 1.0 + st_bass_flash * syn_BassHits;
    if (st_hue != 0.0) { c.rgb = _hueRotate(c.rgb, st_hue); }
    c.a = 1.0;
    return c;
}
#define iResolution strobe_iResolution
#define iTime strobe_iTime
// ---- Shadertoy tab: Image ----
#define SPEED 0.45
#define RINGS (auto_0)
#define BLOOM (auto_1)

vec3 ramp(float t)
{
    return mix(auto_2, auto_3, smoothstep(0.0, 1.0, t));
}

void mainImage( out vec4 fragColor, in vec2 fragCoord )
{
    vec2 p = (2.0 * fragCoord - iResolution.xy) / iResolution.y;
    float r = length(p);
    float a = atan(p.y, p.x);
    float depth = 0.4 / max(r, 0.02) + auto_4;
    float rings = 0.5 + 0.5 * sin(depth * RINGS);
    float spokes = 0.5 + 0.5 * cos(a * 6.0 + auto_5);
    vec3 col = ramp(rings * spokes) * BLOOM * smoothstep(0.0, 0.4, r);
    fragColor = vec4(col, 1.0);
}
// ---- Strobe entry (generated) ----
vec4 renderMain() {
    vec4 strobe_color = vec4(0.0, 0.0, 0.0, 1.0);
    strobe_begin();
    mainImage(strobe_color, strobe_fragCoord);
    return strobe_end(strobe_color);
}
